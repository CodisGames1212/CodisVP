cmake_minimum_required(VERSION 3.22)
project(codisvp LANGUAGES CXX)
file(WRITE "${CMAKE_BINARY_DIR}/.gdignore" "")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GODOT_CPP_DIR "${CMAKE_SOURCE_DIR}/third_party/godot-cpp" CACHE PATH "godot-cpp source")
set(FFMPEG_ROOT "" CACHE PATH "FFmpeg development SDK prefix")
set(GODOTCPP_API_VERSION "4.3" CACHE STRING "Minimum Godot API")
set(GODOTCPP_BUILD_PROFILE "${CMAKE_SOURCE_DIR}/native/build_profile.json" CACHE FILEPATH "Bindings profile")
add_subdirectory("${GODOT_CPP_DIR}" godot-cpp)
find_package(Threads REQUIRED)
add_library(codisvp SHARED native/codisvp.cpp native/decoder.cpp)
target_link_libraries(codisvp PRIVATE godot-cpp Threads::Threads)
target_compile_definitions(codisvp PRIVATE NOMINMAX)
find_path(FFMPEG_INCLUDE libavcodec/avcodec.h HINTS "${FFMPEG_ROOT}/include" REQUIRED)
target_include_directories(codisvp PRIVATE "${FFMPEG_INCLUDE}")
option(CODISVP_STATIC_FFMPEG "Link the CI FFmpeg static SDK and its system dependencies" OFF)
if(CODISVP_STATIC_FFMPEG)
  find_package(PkgConfig REQUIRED)
  set(ENV{PKG_CONFIG_LIBDIR} "${FFMPEG_ROOT}/lib/pkgconfig")
  pkg_check_modules(FFMPEG REQUIRED libavformat libavcodec libavutil libswscale libswresample)
  target_link_directories(codisvp PRIVATE ${FFMPEG_STATIC_LIBRARY_DIRS})
  target_link_libraries(codisvp PRIVATE ${FFMPEG_STATIC_LIBRARIES})
  # Keep repeated Apple '-framework NAME' pairs together during option deduplication.
  string(JOIN " " FFMPEG_SYSTEM_LINK_OPTIONS ${FFMPEG_STATIC_LDFLAGS_OTHER})
  if(FFMPEG_SYSTEM_LINK_OPTIONS)
    target_link_options(codisvp PRIVATE "SHELL:${FFMPEG_SYSTEM_LINK_OPTIONS}")
  endif()
else()
  foreach(component avformat avcodec avutil swscale swresample)
    find_library(FFMPEG_${component} NAMES ${component} HINTS "${FFMPEG_ROOT}/lib" REQUIRED)
    target_link_libraries(codisvp PRIVATE "${FFMPEG_${component}}")
  endforeach()
endif()
if(WIN32)
  set(CODISVP_PLATFORM windows)
elseif(ANDROID)
  set(CODISVP_PLATFORM android)
elseif(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(CODISVP_PLATFORM ios)
elseif(APPLE)
  set(CODISVP_PLATFORM macos)
else()
  set(CODISVP_PLATFORM linux)
endif()
godot_arch_name(CODISVP_ARCH)
set_target_properties(codisvp PROPERTIES
  PREFIX "" OUTPUT_NAME "codisvp.${CODISVP_PLATFORM}.${CODISVP_ARCH}"
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/addons/codisvp/bin"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/addons/codisvp/bin"
  BUILD_RPATH "$ORIGIN;@loader_path" INSTALL_RPATH "$ORIGIN;@loader_path")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(codisvp PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/addons/codisvp/bin" LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/addons/codisvp/bin")
endforeach()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set_target_properties(codisvp PROPERTIES FRAMEWORK TRUE
    FRAMEWORK_VERSION A MACOSX_FRAMEWORK_IDENTIFIER games.codis.codisvp
    OUTPUT_NAME codisvp.ios.arm64 XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO)
endif()
if(WIN32)
  file(GLOB FFMPEG_DLLS "${FFMPEG_ROOT}/bin/*.dll")
  foreach(dll IN LISTS FFMPEG_DLLS)
    add_custom_command(TARGET codisvp POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${dll}" "$<TARGET_FILE_DIR:codisvp>")
  endforeach()
endif()
